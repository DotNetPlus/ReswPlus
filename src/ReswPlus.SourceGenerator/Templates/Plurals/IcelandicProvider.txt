using System;
using _ReswPlus_AutoGenerated.Utils;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class IcelandicProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            // Icelandic declines on the last digit, of the whole number when the quantity carries no decimals
            // and of the decimals themselves when it does: 21 and 0.1 take the singular, 11 and 0.2 do not.
            var decimals = n.DigitsAfterDecimal();

            if (decimals == 0)
            {
                var i = Math.Truncate(n);
                if (i % 10 == 1 && i % 100 != 11)
                {
                    return PluralTypeEnum.ONE;
                }
            }
            else if (decimals % 10 == 1 && decimals % 100 != 11)
            {
                return PluralTypeEnum.ONE;
            }

            return PluralTypeEnum.OTHER;
        }
    }
}
