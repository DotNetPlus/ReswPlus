
namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class OnlyOneOrMillionsProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            if (n == 1)
            {
                return PluralTypeEnum.ONE;
            }
            // 'many' covers the exact non-zero multiples of a million. CLDR also assigns it to compact
            // notations such as "3M", which cannot be expressed here: this API only receives the quantity.
            if (n != 0 && n % 1000000 == 0)
            {
                return PluralTypeEnum.MANY;
            }
            return PluralTypeEnum.OTHER;
        }

    }
}
