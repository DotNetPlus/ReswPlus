using System;
using _ReswPlus_AutoGenerated.Utils;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class SlovenianProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            // 'few' also covers every quantity with visible decimals (v != 0).
            if (!n.IsInt())
            {
                return PluralTypeEnum.FEW;
            }

            // The categories are selected on i % 100, so they repeat every hundred: 101 is 'one' and 103 is
            // 'few', exactly like 1 and 3. The integer part is taken with Math.Truncate rather than with a
            // cast, which would overflow past the 32-bit range.
            var integerModulo = Math.Truncate(n) % 100;

            if (integerModulo == 1)
            {
                return PluralTypeEnum.ONE;
            }
            if (integerModulo == 2)
            {
                return PluralTypeEnum.TWO;
            }
            if (integerModulo == 3 || integerModulo == 4)
            {
                return PluralTypeEnum.FEW;
            }

            return PluralTypeEnum.OTHER;
        }
    }
}
