using _ReswPlus_AutoGenerated.Utils;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class LithuanianProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            // The CLDR ranges below are integer ranges, so a quantity with decimals never matches them and
            // falls through to the 'many' category instead.
            if (n.IsInt())
            {
                if ((n % 10).IsBetween(2, 9) && !(n % 100).IsBetween(11, 19))
                {
                    return PluralTypeEnum.FEW;
                }
                if ((n % 10) == 1 && !(n % 100).IsBetween(11, 19))
                {
                    return PluralTypeEnum.ONE;
                }
            }
            if (n.GetNumberOfDigitsAfterDecimal() != 0)
                return PluralTypeEnum.MANY;
            return PluralTypeEnum.OTHER;

        }
    }
}
