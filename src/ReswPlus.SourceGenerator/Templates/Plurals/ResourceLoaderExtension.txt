using _ReswPlus_AutoGenerated.Utils;
using System.Globalization;
using System;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal static class ResourceLoaderExtension
    {
        private static _ReswPlus_AutoGenerated.Plurals.IPluralProvider _pluralProvider;
        private static string _pluralLanguage;
        private static readonly object _objLock = new object();

        public static string GetPlural(this _ReswPlus_AutoGenerated.ResourceStringProvider resourceStringProvider, string key, double number, bool supportNoneState = false)
        {
            if (supportNoneState && number == 0)
            {
                var noneSentence = GetString(resourceStringProvider, key, "_None");
                if (noneSentence != null)
                {
                    return noneSentence;
                }
            }

            var provider = GetPluralProvider();
            if (provider is null)
            {
                return "";
            }

            // CLDR defines its operands over the absolute value of the quantity, so that a count of -1 selects
            // the same form as a count of 1.
            var pluralType = provider.ComputePlural(Math.Abs(number));
            var selectedSentence = GetString(resourceStringProvider, key, GetSuffix(pluralType));

            // A resource set doesn't have to declare every form its language can select: a language ReswPlus
            // has rules for may be translated without them, and the '_Other' form is the one every set has.
            // Falling back to it keeps a missing form from erasing the text altogether.
            if (selectedSentence is null && pluralType != PluralTypeEnum.OTHER)
            {
                selectedSentence = GetString(resourceStringProvider, key, "_Other");
            }

            return selectedSentence ?? "";
        }

        /// <summary>
        /// Returns the suffix of the resource holding a plural form.
        /// </summary>
        private static string GetSuffix(PluralTypeEnum pluralType)
        {
            switch (pluralType)
            {
                case PluralTypeEnum.ZERO:
                    return "_Zero";
                case PluralTypeEnum.ONE:
                    return "_One";
                case PluralTypeEnum.TWO:
                    return "_Two";
                case PluralTypeEnum.FEW:
                    return "_Few";
                case PluralTypeEnum.MANY:
                    return "_Many";
                default:
                    return "_Other";
            }
        }

        /// <summary>
        /// Looks up one form of a pluralized resource, or returns null when it isn't declared.
        /// </summary>
        private static string GetString(_ReswPlus_AutoGenerated.ResourceStringProvider resourceStringProvider, string key, string suffix)
        {
            try
            {
                var value = resourceStringProvider.GetString(key + suffix);
                return string.IsNullOrEmpty(value) ? null : value;
            }
            catch
            {
                // A missing resource surfaces as an empty string with some resource loaders and as an
                // exception with others, so both mean the form isn't declared.
                return null;
            }
        }

        /// <summary>
        /// Returns the provider holding the plural rules of the language the resources are resolved in.
        /// </summary>
        /// <remarks>
        /// The provider is kept between calls and rebuilt when the language changes, which an app can do while
        /// it runs by setting <c>ApplicationLanguages.PrimaryLanguageOverride</c>.
        /// </remarks>
        private static _ReswPlus_AutoGenerated.Plurals.IPluralProvider GetPluralProvider(string forcedCultureName = null)
        {
            var language = ResolveLanguage(forcedCultureName);

            lock (_objLock)
            {
                if (_pluralProvider is null || !string.Equals(_pluralLanguage, language, StringComparison.Ordinal))
                {
                    _pluralProvider = GetPluralChooser(language);
                    _pluralLanguage = language;
                }

                return _pluralProvider;
            }
        }

        /// <summary>
        /// Returns the language whose plural rules should be used.
        /// </summary>
        private static string ResolveLanguage(string forcedCultureName)
        {
            var forcedLanguage = ReadLanguage(forcedCultureName);
            if (forcedLanguage != null)
            {
                return forcedLanguage;
            }

            // The language of the app is only resolved when there is no forced culture to use instead.
            return GetPluralLanguage();
        }

        /// Returns the language of the app.
        /// </summary>
        private static string GetPluralLanguage()
        {
{{PluralLanguageResolver}}
        }

        /// <summary>
        /// Reads the language tag an app or a culture names itself with, in the form the selector keys by.
        /// </summary>
        private static string ReadLanguage(string languageTag)
        {
            if (string.IsNullOrEmpty(languageTag))
            {
                return null;
            }

            // The whole tag is kept, because a region can decline differently from the language it belongs to.
            // CultureInfo is not used to read it: it rejects tags such as 'sl-rozaj-biske-1994', which are
            // valid and which the selector handles by falling back on the language.
            return languageTag.Replace('_', '-').ToLowerInvariant();
        }

        /// <summary>
        /// Returns the provider holding the rules of a language tag.
        /// </summary>
        /// <remarks>
        /// The tag is matched whole first and then shortened one subtag at a time, so that 'pt-PT' takes the
        /// rules CLDR gives it while 'pt-BR' falls back on the ones it gives 'pt', and a tag nothing is held
        /// for -- 'fr-CA' -- lands on the rules of its language without having to be listed.
        /// </remarks>
        private static _ReswPlus_AutoGenerated.Plurals.IPluralProvider GetPluralChooser(string culture)
        {
            while (!string.IsNullOrEmpty(culture))
            {

                }

                var separator = culture.LastIndexOf('-');
                culture = separator <= 0 ? null : culture.Substring(0, separator);
            }

            return new _ReswPlus_AutoGenerated.Plurals.OtherProvider();
        }
