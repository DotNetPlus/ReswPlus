using System;
using _ReswPlus_AutoGenerated.Utils;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class SinhalaProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            // 'n = 0,1 or i = 0 and f = 1': a quantity below one whose decimals read as a single one is
            // declined like a single item, but 1.1 is not.
            if (n == 0 || n == 1 || (Math.Truncate(n) == 0 && n.DigitsAfterDecimal() == 1))
            {
                return PluralTypeEnum.ONE;
            }
            return PluralTypeEnum.OTHER;
        }
    }
}
