using System;
using _ReswPlus_AutoGenerated.Utils;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class HebrewProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            var i = Math.Truncate(n);

            if (n.IsInt())
            {
                if (i == 1)
                {
                    return PluralTypeEnum.ONE;
                }
                if (i == 2)
                {
                    return PluralTypeEnum.TWO;
                }
            }
            else if (i == 0)
            {
                // A quantity below one is declined like a single item: 'i = 0 and v != 0'.
                return PluralTypeEnum.ONE;
            }

            return PluralTypeEnum.OTHER;

        }
    }
}
