using _ReswPlus_AutoGenerated.Utils;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class LatvianProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            var v = n.GetNumberOfDigitsAfterDecimal();
            var f = n.DigitsAfterDecimal();

            // 'zero' covers every multiple of ten, not just zero itself, and the whole 11 to 19 range. The
            // CLDR ranges only hold integers, so a quantity such as 11.5 is not part of them.
            if (n % 10 == 0 ||
                ((n % 100).IsInt() && (n % 100).IsBetween(11, 19)) ||
                (v == 2 && (f % 100) >= 11 && (f % 100) <= 19))
            {
                return PluralTypeEnum.ZERO;
            }

            if ((n % 10 == 1 && n % 100 != 11) ||
                (v == 2 && f % 10 == 1 && f % 100 != 11) ||
                (v != 2 && f % 10 == 1))
            {
                return PluralTypeEnum.ONE;
            }

            return PluralTypeEnum.OTHER;
        }
    }
}
