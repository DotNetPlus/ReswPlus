using System;
using _ReswPlus_AutoGenerated.Utils;

namespace _ReswPlus_AutoGenerated.Plurals
{
    internal sealed class DanishProvider : IPluralProvider
    {
        public PluralTypeEnum ComputePlural(double n)
        {
            // 'one' is n = 1, plus every quantity with decimals whose integer part is 0 or 1: 1.5 is 'one'
            // even though it is greater than 1.
            if (n == 1 || (!n.IsInt() && Math.Truncate(n).IsBetween(0, 1)))
                return PluralTypeEnum.ONE;
            return PluralTypeEnum.OTHER;
        }
    }
}
