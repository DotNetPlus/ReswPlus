using System;
using System.Globalization;

namespace _ReswPlus_AutoGenerated.Utils
{
    internal static class DoubleExtension
    {
        /// <summary>
        /// The general formats, by rising number of significant digits, that a quantity is searched through.
        /// </summary>
        private static readonly string[] GeneralFormats =
        {
            "G1", "G2", "G3", "G4", "G5", "G6", "G7", "G8",
            "G9", "G10", "G11", "G12", "G13", "G14", "G15"
        };

        public static bool IsBetween(this double number, double start, double end)
        {
            return start <= number && number <= end;
        }
        public static bool IsBetweenEndNotIncluded(this double number, double start, double end)
        {
            return start <= number && number < end;
        }

        public static bool IsInt(this double number)
        {
            // Truncating rather than casting to an int, which would report every value outside of the 32-bit
            // range as fractional.
            return Math.Truncate(number) == number;
        }

        /// <summary>
        /// Returns the CLDR 'v' operand: the number of visible decimals of the quantity.
        /// </summary>
        public static uint GetNumberOfDigitsAfterDecimal(this double number)
        {
            return (uint)VisibleDecimals(number).Length;
        }

        /// <summary>
        /// Returns the CLDR 'f' operand: the visible decimals of the quantity, read as an integer.
        /// </summary>
        public static long DigitsAfterDecimal(this double number)
        {
            var decimals = VisibleDecimals(number);
            long result;
            if (decimals.Length != 0 && long.TryParse(decimals, NumberStyles.None, CultureInfo.InvariantCulture, out result))
            {
                return result;
            }
            return 0;
        }

        /// <summary>
        /// Returns the decimals of the quantity, or an empty string when it has none.
        /// </summary>
        private static string VisibleDecimals(double number)
        {
            // The decimals are read off the quantity itself rather than off 'number - (int)number': the
            // subtraction turns a value such as 1.4 into 0.3999999999999999, which reports the digits of the
            // binary representation instead of the ones the caller wrote.
            var shortest = ShortestRoundTrip(Math.Abs(number));
            var mantissaAndExponent = shortest.Split('E');
            var mantissa = mantissaAndExponent[0].Split('.');
            var integerDigits = mantissa[0];
            var decimalDigits = mantissa.Length > 1 ? mantissa[1] : "";

            if (mantissaAndExponent.Length == 1)
            {
                return decimalDigits;
            }

            // Small and large quantities come back in scientific notation, which has to be expanded before its
            // decimals can be read: those of '1E-06' are five zeros and a one, not none at all.
            var exponent = int.Parse(mantissaAndExponent[1], NumberStyles.AllowLeadingSign, CultureInfo.InvariantCulture);
            var digits = integerDigits + decimalDigits;
            var pointPosition = integerDigits.Length + exponent;

            if (pointPosition >= digits.Length)
            {
                return "";
            }
            if (pointPosition <= 0)
            {
                return new string('0', -pointPosition) + digits;
            }
            return digits.Substring(pointPosition);
        }

        /// <summary>
        /// Returns the shortest representation of a quantity that reads back as the same quantity.
        /// </summary>
        /// <remarks>
        /// Which decimals a quantity has is what CLDR selects a plural form on, and the shortest
        /// representation that round-trips is the closest a double gets to the ones the caller wrote. It is
        /// searched for rather than being asked of the 'R' format, whose result differs between .NET and .NET
        /// Framework, which would have the same quantity select different forms in a UWP and in a WinAppSDK
        /// app. The search stops at fifteen digits because that is as far as the two agree on reading a
        /// quantity back; past it they can pick different representations of a value that is carrying the
        /// noise of its binary form rather than digits anyone wrote.
        /// </remarks>
        private static string ShortestRoundTrip(double number)
        {
            if (double.IsNaN(number) || double.IsInfinity(number))
            {
                return "0";
            }

            for (var index = 0; index < GeneralFormats.Length; index++)
            {
                var candidate = number.ToString(GeneralFormats[index], CultureInfo.InvariantCulture);
                double parsed;
                if (double.TryParse(candidate, NumberStyles.Float, CultureInfo.InvariantCulture, out parsed) &&
                    parsed == number)
                {
                    return candidate;
                }
            }

            return number.ToString("R", CultureInfo.InvariantCulture);
        }
    }
}
