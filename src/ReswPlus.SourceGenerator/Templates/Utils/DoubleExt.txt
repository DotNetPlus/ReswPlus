using System;
using System.Globalization;

namespace _ReswPlus_AutoGenerated.Utils
{
    internal static class DoubleExtension
    {
        /// <summary>
        /// A format that always writes a quantity in fixed point, with every decimal a double can hold.
        /// </summary>
        private const string FixedPointFormat = "0.#####################";

        public static bool IsBetween(this double number, double start, double end)
        {
            return start <= number && number <= end;
        }
        public static bool IsBetweenEndNotIncluded(this double number, double start, double end)
        {
            return start <= number && number < end;
        }

        public static bool IsInt(this double number)
        {
            // Truncating rather than casting to an int, which would report every value outside of the 32-bit
            // range as fractional.
            return Math.Truncate(number) == number;
        }

        /// <summary>
        /// Returns the CLDR 'v' operand: the number of visible decimals of the quantity.
        /// </summary>
        public static uint GetNumberOfDigitsAfterDecimal(this double number)
        {
            return (uint)VisibleDecimals(number).Length;
        }

        /// <summary>
        /// Returns the CLDR 'f' operand: the visible decimals of the quantity, read as an integer.
        /// </summary>
        public static long DigitsAfterDecimal(this double number)
        {
            var decimals = VisibleDecimals(number);
            long result;
            if (decimals.Length != 0 && long.TryParse(decimals, NumberStyles.None, CultureInfo.InvariantCulture, out result))
            {
                return result;
            }
            return 0;
        }

        /// <summary>
        /// Returns the decimals of the quantity, or an empty string when it has none.
        /// </summary>
        private static string VisibleDecimals(double number)
        {
            // The decimals are read off the quantity itself rather than off 'number - (int)number': the
            // subtraction turns a value such as 1.4 into 0.3999999999999999, which reports the digits of the
            // binary representation instead of the ones the caller wrote.
            // The format is spelled out because the default one switches to scientific notation for small and
            // large values, where '1E-06' has no decimals to read and '1.2E+20' has '2E+20' as its decimals.
            var parts = Math.Abs(number).ToString(FixedPointFormat, CultureInfo.InvariantCulture).Split('.');
            return parts.Length > 1 ? parts[1] : "";
        }
    }
}
